# The reflective constructors of FfmpegAudioRenderer / Libdav1dVideoRenderer are already kept by
# media3-exoplayer's own consumer rules; these cover the JNI side.

    native <methods>;
}

}
-keep class androidx.media3.decoder.DecoderInputBuffer {
  *;
}
-keep class androidx.media3.decoder.av1.Dav1dDecoder {
  *;
}
